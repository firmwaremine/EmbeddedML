cmake_minimum_required(VERSION 3.16)
project(eml_perception_labs LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(perception STATIC perception.c)
target_include_directories(perception PUBLIC .)
target_compile_options(perception PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(perception_test unit.c)
target_compile_options(perception_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(perception_test PRIVATE perception m)
enable_testing()
add_test(NAME perception_boundaries COMMAND perception_test)
add_library(perception_shared SHARED perception.c)
set_target_properties(perception_shared PROPERTIES OUTPUT_NAME perception)
target_compile_options(perception_shared PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(perception_shared PRIVATE m)
add_executable(policy_trace policy_trace.c)
target_compile_options(policy_trace PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(policy_trace PRIVATE perception m)
