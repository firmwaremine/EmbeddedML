cmake_minimum_required(VERSION 3.16)
project(eml_pure_c LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(eml_c STATIC eml.c stream.c)
target_include_directories(eml_c PUBLIC .)
target_compile_options(eml_c PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(eml_c_cli cli.c)
target_link_libraries(eml_c_cli PRIVATE eml_c m)
target_compile_options(eml_c_cli PRIVATE -Wall -Wextra -Wpedantic -Werror)

enable_testing()
add_executable(eml_c_unit stream_test.c)
target_link_libraries(eml_c_unit PRIVATE eml_c m)
target_compile_options(eml_c_unit PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME pure_c_boundaries COMMAND eml_c_unit)
