cmake_minimum_required(VERSION 3.16)
project(eml_system_labs LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(bounded STATIC bounded.c)
target_include_directories(bounded PUBLIC .)
target_compile_options(bounded PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(bounded_test unit.c)
target_compile_options(bounded_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(bounded_test PRIVATE bounded m)
enable_testing()
add_test(NAME bounded_adaptation COMMAND bounded_test)
