cmake_minimum_required(VERSION 3.16)
project(eml_sensing_labs LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(eml_sensing STATIC sensing.c)
target_include_directories(eml_sensing PUBLIC .)
target_compile_options(eml_sensing PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(sensing_cli lab_cli.c)
target_link_libraries(sensing_cli PRIVATE eml_sensing m)
target_compile_options(sensing_cli PRIVATE -Wall -Wextra -Wpedantic -Werror)
